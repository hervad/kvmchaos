# kvmchaos

Agent-less chaos engineering CLI for KVM/libvirt virtual machines.

Injects reversible faults (`vm.pause`, `vm.kill`) into VMs from the hypervisor
layer via libvirt. No guest-side agent. Linux host only. **Lab use only.**

## Prerequisites

- Fedora / RHEL / Debian-based Linux with libvirt installed and `libvirtd` running.
- Python 3.14+.
- System packages (Fedora): `sudo dnf install libvirt-devel pkg-config`.
- User in the `libvirt` group: `sudo usermod -aG libvirt $USER` (re-login).
- `uv` installed: see https://docs.astral.sh/uv/.

## Install (from source)

```bash
git clone <repo> kvmchaos && cd kvmchaos
uv sync
```

## Usage

```bash
uv run kvmchaos --version
uv run kvmchaos list-vms
uv run kvmchaos list-faults
uv run kvmchaos inject vm.pause <vm-name>
uv run kvmchaos inject vm.kill  <vm-name> --yes
```

Connection URI precedence: `--connect` flag > `$LIBVIRT_DEFAULT_URI` > `qemu:///system`.

## Warnings

- `vm.pause` freezes vCPUs but keeps RAM. Guest clocks drift.
- `vm.kill` destroys the running guest gracefully (SIGTERM then SIGKILL).
  Unsaved guest state is lost. `revert()` starts the VM from its defined config.
- This tool is **not** production-safe. Do not point it at anything you care about.

## Development

```bash
uv run pytest
uv run ruff check
uv run ruff format --check
uv run ty check
```

## License

MIT.
```

```python
"""Libvirt URI resolution and connection management.

The URI is resolved at call time with the precedence documented by `virsh`:
explicit argument → `LIBVIRT_DEFAULT_URI` env var → hardcoded default.
Keeping this in one place means the CLI, tests, and library callers all
agree on which hypervisor they are talking to.
"""

from __future__ import annotations

import os
from collections.abc import Iterator
from contextlib import contextmanager

import libvirt

DEFAULT_URI = "qemu:///system"


def resolve_uri(cli_flag: str | None) -> str:
    """Resolve the libvirt URI using CLI > env > default precedence.

    Args:
        cli_flag: Value of `--connect` if passed, else None.

    Returns:
        The URI string to pass to `libvirt.open`.
    """
    if cli_flag:
        return cli_flag
    env = os.environ.get("LIBVIRT_DEFAULT_URI")
    if env:
        return env
    return DEFAULT_URI


@contextmanager
def connect(uri: str | None = None) -> Iterator[libvirt.virConnect]:
    """Open a libvirt connection and close it on exit.

    Args:
        uri: Optional explicit URI. If None, `resolve_uri` is consulted.

    Yields:
        A live `libvirt.virConnect`. Closed automatically when the `with`
        block exits, including on exception.

    Raises:
        libvirt.libvirtError: If the connection cannot be opened.
    """
    resolved = resolve_uri(uri)
    conn = libvirt.open(resolved)
    try:
        yield conn
    finally:
        # libvirt raises if close() is called twice; we trust the caller
        # not to close it inside the block.
        conn.close()
```

```python
"""Append-only JSONL event log for kvmchaos runs.

Events are written via the stdlib `logging` framework so callers never
touch a file handle directly. One line per event, JSON-encoded, UTF-8.
"""

from __future__ import annotations

import json
import logging
import os
from datetime import UTC, datetime
from logging.handlers import RotatingFileHandler
from pathlib import Path

_LOGGER_NAME = "kvmchaos.events"
_MAX_BYTES = 10 * 1024 * 1024  # 10 MiB before rotation
_BACKUP_COUNT = 3


class JsonFormatter(logging.Formatter):
    """Formats `LogRecord.msg` (a dict) as a single JSON line."""

    def format(self, record: logging.LogRecord) -> str:
        payload: dict[str, object] = {
            "ts": datetime.now(UTC).isoformat(timespec="seconds"),
        }
        # record.msg is the event dict passed to log_event()
        if isinstance(record.msg, dict):
            payload.update(record.msg)
        else:
            payload["message"] = record.getMessage()
        return json.dumps(payload, separators=(",", ":"), sort_keys=False)


def default_log_path() -> Path:
    """Return the XDG state path for the event log."""
    base = os.environ.get("XDG_STATE_HOME") or str(Path.home() / ".local" / "state")
    return Path(base) / "kvmchaos" / "events.log"


def configure_logging(path: Path | None = None) -> None:
    """Attach a rotating JSONL handler to the `kvmchaos.events` logger.

    Idempotent: repeated calls do not add duplicate handlers.

    Args:
        path: Target log file. Defaults to `default_log_path()`.
    """
    target = path or default_log_path()
    target.parent.mkdir(parents=True, exist_ok=True)

    logger = logging.getLogger(_LOGGER_NAME)
    logger.setLevel(logging.INFO)
    logger.propagate = False

    if logger.handlers:
        return  # already configured

    handler = RotatingFileHandler(
        target, maxBytes=_MAX_BYTES, backupCount=_BACKUP_COUNT, encoding="utf-8"
    )
    handler.setFormatter(JsonFormatter())
    logger.addHandler(handler)


def log_event(**fields: object) -> None:
    """Write a single JSON event line.

    Keyword arguments become top-level keys in the JSON line. A `ts`
    field is added automatically. Common keys: `action`, `fault`, `vm`,
    `result`, `duration_ms`, `error`.
    """
    logging.getLogger(_LOGGER_NAME).info(fields)
```

"""Interactive confirmation helper."""

from __future__ import annotations

import sys


def confirm(prompt: str, *, assume_yes: bool) -> bool:
    """Ask the user to confirm an action.

    Args:
        prompt: Human-readable question to display.
        assume_yes: If True, skip the prompt and return True immediately.

    Returns:
        True if the user typed `y` or `yes` (any case), else False.
        Empty input defaults to False.
    """
    if assume_yes:
        return True
    print(f"{prompt} [y/N] ", end="", flush=True)
    answer = sys.stdin.readline().strip().lower()
    return answer in {"y", "yes"}
```

```python
"""Structural type for chaos faults.

A fault is a stateless triple of operations against a libvirt domain:

    inject -> verify -> revert

`verify` confirms the fault actually took effect (not merely that the
libvirt call returned). `revert` returns the domain to its pre-inject
state. Faults that cannot be reverted belong in a different module.
"""

from __future__ import annotations

from typing import ClassVar, Protocol

import libvirt


class Fault(Protocol):
    """Structural interface all faults must satisfy."""

    name: ClassVar[str]
    """Dotted identifier, e.g. 'vm.pause'. Used as the registry key."""

    description: ClassVar[str]
    """One-line human summary shown by `kvmchaos list-faults`."""

    destructive: ClassVar[bool]
    """True if the fault kills or otherwise disrupts guest state."""

    def inject(self, domain: libvirt.virDomain) -> None:
        """Apply the fault. Raises on libvirt error."""
        ...

    def verify(self, domain: libvirt.virDomain) -> None:
        """Assert the fault took effect. Raises `RuntimeError` otherwise."""
        ...

    def revert(self, domain: libvirt.virDomain) -> None:
        """Undo the fault. Raises on libvirt error."""
        ...
```

"""`vm.pause` — suspend a running domain's vCPUs."""

from __future__ import annotations

from typing import ClassVar

import libvirt


class VmPauseFault:
    """Pauses a running VM via `virDomain.suspend()`.

    Freezes vCPU execution while keeping RAM contents. The guest does not
    observe the pause directly, but its clock will drift relative to
    wall time. Reverted by `virDomain.resume()`.
    """

    name: ClassVar[str] = "vm.pause"
    description: ClassVar[str] = "Suspend vCPUs; RAM preserved. Clock drifts."
    destructive: ClassVar[bool] = False

    def inject(self, domain: libvirt.virDomain) -> None:
        """Suspend the domain."""
        domain.suspend()

    def verify(self, domain: libvirt.virDomain) -> None:
        """Assert the domain is in the paused state.

        Raises:
            RuntimeError: if the domain state is anything other than
                `VIR_DOMAIN_PAUSED`.
        """
        state, _ = domain.state()
        if state != libvirt.VIR_DOMAIN_PAUSED:
            raise RuntimeError(
                f"domain '{domain.name()}' not paused after inject "
                f"(state={state})"
            )

    def revert(self, domain: libvirt.virDomain) -> None:
        """Resume the domain."""
        domain.resume()
```

"""`vm.kill` — forcefully terminate a running domain."""

from __future__ import annotations

from typing import ClassVar

import libvirt


class VmKillFault:
    """Destroys a running VM, then restarts it from its defined config.

    Uses `VIR_DOMAIN_DESTROY_GRACEFUL` which sends SIGTERM first and only
    escalates to SIGKILL on timeout. Any unsaved guest state is lost.
    Revert calls `create()`, which only works if the domain is
    persistently defined (the usual case for `qemu:///system`).
    """

    name: ClassVar[str] = "vm.kill"
    description: ClassVar[str] = "Graceful destroy then restart. Unsaved state lost."
    destructive: ClassVar[bool] = True

    def inject(self, domain: libvirt.virDomain) -> None:
        """Destroy the domain (graceful-first)."""
        domain.destroyFlags(libvirt.VIR_DOMAIN_DESTROY_GRACEFUL)

    def verify(self, domain: libvirt.virDomain) -> None:
        """Assert the domain is in the shut-off state.

        Raises:
            RuntimeError: if the state is not `VIR_DOMAIN_SHUTOFF`.
        """
        state, _ = domain.state()
        if state != libvirt.VIR_DOMAIN_SHUTOFF:
            raise RuntimeError(
                f"domain '{domain.name()}' not shut off after inject "
                f"(state={state})"
            )

    def revert(self, domain: libvirt.virDomain) -> None:
        """Start the domain from its persisted definition."""
        domain.create()
```

"""Typer entry point for kvmchaos.

The CLI is a thin orchestration layer: it resolves a libvirt URI, looks
up a fault in the registry, prompts for confirmation, and drives the
inject/verify/revert sequence while logging each step.
"""

from __future__ import annotations

import typer

from kvmchaos import __version__
from kvmchaos.faults import FAULTS
from kvmchaos.libvirt_conn import connect

app = typer.Typer(
    help="Agent-less chaos engineering for KVM/libvirt VMs.",
    no_args_is_help=True,
    add_completion=False,
)


def _version_callback(value: bool) -> None:
    if value:
        typer.echo(f"kvmchaos {__version__}")
        raise typer.Exit()


@app.callback()
def _root(
    version: bool = typer.Option(
        False, "--version", callback=_version_callback, is_eager=True,
        help="Show version and exit.",
    ),
    connect_uri: str | None = typer.Option(
        None, "--connect", help="libvirt URI (overrides LIBVIRT_DEFAULT_URI).",
    ),
    ctx: typer.Context = typer.Context,  # type: ignore[assignment]
) -> None:
    """Root callback stashes the connect URI on the Typer context."""
    if isinstance(ctx, typer.Context):
        ctx.obj = {"connect_uri": connect_uri}


@app.command("list-vms")
def list_vms(ctx: typer.Context) -> None:
    """List domains on the connected libvirt URI."""
    uri = ctx.obj.get("connect_uri") if ctx.obj else None
    with connect(uri) as conn:
        for domain in conn.listAllDomains():
            state, _ = domain.state()
            typer.echo(f"{domain.name()}\t{_state_name(state)}")


@app.command("list-faults")
def list_faults() -> None:
    """List registered faults with their descriptions."""
    for name, fault in FAULTS.items():
        marker = "destructive" if fault.destructive else "safe"
        typer.echo(f"{name}\t[{marker}]\t{fault.description}")


_STATE_NAMES = {
    0: "nostate",
    1: "running",
    2: "blocked",
    3: "paused",
    4: "shutdown",
    5: "shutoff",
    6: "crashed",
    7: "suspended",
}


def _state_name(state: int) -> str:
    return _STATE_NAMES.get(state, f"state{state}")
```

import time

import libvirt

from kvmchaos.eventlog import configure_logging, log_event
from kvmchaos.safety import confirm


@app.command("inject")
def inject_cmd(
    ctx: typer.Context,
    fault_name: str = typer.Argument(..., metavar="FAULT", help="Fault name (see list-faults)."),
    vm: str = typer.Argument(..., metavar="VM", help="Target VM name."),
    assume_yes: bool = typer.Option(False, "--yes", "-y", help="Skip confirmation prompt."),
) -> None:
    """Inject a fault into a VM, verify it, then revert."""
    configure_logging()

    if fault_name not in FAULTS:
        typer.echo(
            f"unknown fault: {fault_name!r}. "
            f"Known: {', '.join(sorted(FAULTS))}",
            err=True,
        )
        raise typer.Exit(code=2)
    fault = FAULTS[fault_name]

    uri = ctx.obj.get("connect_uri") if ctx.obj else None
    with connect(uri) as conn:
        try:
            domain = conn.lookupByName(vm)
        except libvirt.libvirtError as e:
            typer.echo(f"VM '{vm}' not found: {e}", err=True)
            raise typer.Exit(code=2) from e

        prompt = (
            f"About to inject '{fault_name}' on VM '{vm}' "
            f"({'DESTRUCTIVE' if fault.destructive else 'reversible'}). Continue?"
        )
        if not confirm(prompt, assume_yes=assume_yes):
            typer.echo("aborted.")
            raise typer.Exit(code=1)

        _run_step(fault.inject, domain, action="inject", fault_name=fault_name, vm=vm)
        try:
            _run_step(fault.verify, domain, action="verify", fault_name=fault_name, vm=vm)
        except Exception:
            # Verify failed — still attempt revert so we don't leave the VM broken.
            _run_step(
                fault.revert, domain, action="revert", fault_name=fault_name, vm=vm,
                swallow=True,
            )
            raise typer.Exit(code=1)
        _run_step(fault.revert, domain, action="revert", fault_name=fault_name, vm=vm)


def _run_step(
    func,
    domain: libvirt.virDomain,
    *,
    action: str,
    fault_name: str,
    vm: str,
    swallow: bool = False,
) -> None:
    """Run one fault step, log the outcome, re-raise on error unless swallowed."""
    t0 = time.monotonic()
    try:
        func(domain)
    except Exception as exc:  # noqa: BLE001 — we re-log and re-raise
        log_event(
            action=action, fault=fault_name, vm=vm, result="fail",
            duration_ms=int((time.monotonic() - t0) * 1000), error=str(exc),
        )
        if swallow:
            typer.echo(f"{action} failed: {exc}", err=True)
            return
        typer.echo(f"{action} failed: {exc}", err=True)
        raise
    log_event(
        action=action, fault=fault_name, vm=vm, result="ok",
        duration_ms=int((time.monotonic() - t0) * 1000),
    )
```
